# requirements.txt = the libraries the DEPLOYED API needs.
# Render installs ONLY this file (build command: pip install -r requirements.txt).
#
# WHY keep it small: Render's free tier has only 512 MB of RAM. Every heavy library
# we leave out of this file (torch, transformers, sentence-transformers) is memory
# we save in production.

fastapi         # the web framework for our API (/health, /route, /answer)
uvicorn         # the web server that runs the FastAPI app
scikit-learn    # TF-IDF + Logistic Regression (our deployed intent model)
nltk            # Porter stemming, the first step of the deployed model (pure Python, light)
numpy           # fast maths on arrays (lists of numbers)
faiss-cpu       # vector search for the FAQ answers (RAG)
openai          # LLM client (works with OpenAI, Gemini and Ollama via their OpenAI-style APIs)
python-dotenv   # reads secrets from a local .env file (never committed)
sqlalchemy      # talks to the database where we log every request
psycopg[binary] # the PostgreSQL driver (Render's database is PostgreSQL)
